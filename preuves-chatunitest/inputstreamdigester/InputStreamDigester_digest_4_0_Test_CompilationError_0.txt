@ExtendWith(MockitoExtension.class)
public class InputStreamDigester_digest_4_0_Test {

    @InjectMocks
    private InputStreamDigester inputStreamDigester;

    @Mock
    private Encoder encoder;

    @Mock
    private ParseContext parseContext;

    @Mock
    private CacheMemoryBudget cacheMemoryBudget;

    @Mock
    private TikaInputStream tikaInputStream;

    @BeforeEach
    public void setUp() {
        inputStreamDigester = new InputStreamDigester("SHA-256", "sha256Digest", encoder);
        tikaInputStream = new TikaInputStream(null, new TemporaryResources(), null);
        tikaInputStream.setOpenContainer(tikaInputStream);
    }

    @Test
    public void testDigest() throws IOException, NoSuchAlgorithmException {
        when(tikaInputStream.read(any(byte[].class))).thenReturn(10, -1);
        when(parseContext.get(CacheMemoryBudget.class)).thenReturn(cacheMemoryBudget);
        inputStreamDigester.digest(tikaInputStream, new Metadata(), parseContext);
        verify(tikaInputStream).enableRewind(cacheMemoryBudget);
        verify(tikaInputStream).tryRetainInMemory();
        verify(tikaInputStream).rewind();
        verify(encoder).encode(any(byte[].class));
        verify(tikaInputStream).seekTo(0);
        verify(tikaInputStream).setLength(anyLong());
        verify(tikaInputStream).set(any(String.class));
        verify(tikaInputStream).set(any(Property.class), any(String.class));
        verify(tikaInputStream).set(any(Property.class), any(String[].class));
        verify(tikaInputStream).set(any(Property.class), any(int.class));
        verify(tikaInputStream).set(any(Property.class), any(long.class));
        verify(tikaInputStream).set(any(Property.class), any(boolean.class));
        verify(tikaInputStream).set(any(Property.class), any(double.class));
        verify(tikaInputStream).set(any(Property.class), any(Date.class));
        verify(tikaInputStream).set(any(Property.class), any(Calendar.class));
    }
}
