public class InputStreamDigester_digest_4_1_Test {

    private InputStreamDigester digester;

    private TikaInputStream tis;

    private Metadata metadata;

    private ParseContext parseContext;

    private Encoder encoder;

    @BeforeEach
    public void setUp() throws IOException {
        // Mock dependencies
        encoder = mock(Encoder.class);
        parseContext = mock(ParseContext.class);
        tis = mock(TikaInputStream.class);
        metadata = mock(Metadata.class);
        // Initialize the digester
        digester = new InputStreamDigester("SHA-256", "digest", encoder);
    }

    @Test
    public void testDigestValidInput() throws IOException, NoSuchAlgorithmException {
        // Mock the behavior of TikaInputStream
        when(tis.read(any(byte[].class))).thenReturn(10, 20, 30, -1);
        when(tis.getLength()).thenReturn(60L);
        // Call the digest method
        digester.digest(tis, metadata, parseContext);
        // Verify the behavior of the digester
        verify(encoder).encode(argThat(this::isSha256Digest));
        verify(metadata).set(eq("digest"), argThat(this::isSha256Digest));
        verify(metadata).set(eq(TikaCoreProperties.CONTENT_LENGTH), eq("60"));
    }

    @Test
    public void testDigestInvalidInput() throws IOException, NoSuchAlgorithmException {
        // Mock the behavior of TikaInputStream
        when(tis.read(any(byte[].class))).thenReturn(-1);
        when(tis.getLength()).thenReturn(-1);
        // Call the digest method
        digester.digest(tis, metadata, parseContext);
        // Verify the behavior of the digester
        verify(encoder, never()).encode(any(byte[].class));
        verify(metadata, never()).set(eq("digest"), any(byte[].class));
        verify(metadata, never()).set(eq(TikaCoreProperties.CONTENT_LENGTH), eq("60"));
    }

    @Test
    public void testDigestNoContentLength() throws IOException, NoSuchAlgorithmException {
        // Mock the behavior of TikaInputStream
        when(tis.read(any(byte[].class))).thenReturn(10, 20, 30, -1);
        when(tis.getLength()).thenReturn(-1);
        // Call the digest method
        digester.digest(tis, metadata, parseContext);
        // Verify the behavior of the digester
        verify(encoder).encode(argThat(this::isSha256Digest));
        verify(metadata).set(eq("digest"), argThat(this::isSha256Digest));
        verify(metadata, never()).set(eq(TikaCoreProperties.CONTENT_LENGTH), eq("60"));
    }

    private boolean isSha256Digest(byte[] digest) {
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            return Arrays.equals(md.digest(), digest);
        } catch (NoSuchAlgorithmException e) {
            return false;
        }
    }
}
