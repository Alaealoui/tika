@ExtendWith(MockitoExtension.class)
public class InputStreamDigester_digestSink_5_1_Test {

    @Mock
    private Encoder encoder;

    @Mock
    private ParseContext parseContext;

    @Mock
    private Provider provider;

    private InputStreamDigester inputStreamDigester;

    @BeforeEach
    public void setUp() {
        MockitoAnnotations.openMocks(this);
        inputStreamDigester = new InputStreamDigester("SHA-256", "customDigest", encoder);
    }

    @Test
    public void testDigestSink() throws IOException, NoSuchAlgorithmException {
        // Create a Metadata object
        Metadata metadata = new Metadata();
        metadata.set(HttpHeaders.CONTENT_LENGTH, "1234");
        // Create a MessageDigest object
        MessageDigest messageDigest = mock(MessageDigest.class);
        when(messageDigest.digest()).thenReturn(new byte[] { 1, 2, 3 });
        // Create a DigestSink object using reflection
        DigestSink digestSink = (DigestSink) invokePrivate(inputStreamDigester, "digestSink", metadata, parseContext);
        // Write data to the DigestSink
        digestSink.write(1);
        digestSink.write(new byte[] { 2, 3, 4 });
        // Finish the DigestSink
        digestSink.finish(true);
        // Verify that the messageDigest was updated correctly
        verify(messageDigest).update((byte) 1);
        verify(messageDigest).update((byte) 2);
        verify(messageDigest).update((byte) 3);
        verify(messageDigest).update((byte) 2);
        verify(messageDigest).update((byte) 3);
        verify(messageDigest).update((byte) 4);
        // Verify that the total length was updated correctly
        assertEquals(6L, digestSink.getTotal());
        // Verify that the metadata was updated correctly
        assertEquals("8c1a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a8a", metadata.get(inputStreamDigester.metadataProperty.getName()));
    }

    private Object invokePrivate(Object obj, String methodName, Object... args) throws Exception {
        Class<?> clazz = obj.getClass();
        java.lang.reflect.Method method = clazz.getDeclaredMethod(methodName, args.getClass());
        method.setAccessible(true);
        return method.invoke(obj, args);
    }

        method.setAccessible(true);
        return method.invoke(obj, args);
    }
}
