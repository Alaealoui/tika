@ExtendWith(MockitoExtension.class)
public class InputStreamDigester_digestSink_5_0_Test {

    private InputStreamDigester digester;

    private Metadata metadata;

    private ParseContext parseContext;

    private Encoder encoder;

    @BeforeEach
    public void setUp() {
        metadata = new Metadata();
        parseContext = new ParseContext();
        encoder = mock(Encoder.class);
        digester = new InputStreamDigester("SHA-256", "digest", encoder);
    }

    @Test
    public void testDigestSink() throws IOException {
        // Create a mock OutputStream for testing
        OutputStream mockOutputStream = mock(OutputStream.class);
        // Create a DigestSink instance
        DigestSink sink = digester.digestSink(metadata, parseContext);
        // Write some data to the sink
        sink.write(1, new byte[] { 1, 2, 3 });
        sink.write(4, new byte[] { 4, 5, 6 });
        // Finish the sink
        sink.finish(true);
        // Verify that the digest was correctly set in the metadata
        String expectedDigest = "sha256-1234567890abcdef1234567890abcdef1234567890abcdef1234567890abcdef";
        assertEquals(expectedDigest, metadata.get(digester.metadataProperty.getName()));
        // Verify that the content length was set in the metadata
        assertEquals("6", metadata.get(HttpHeaders.CONTENT_LENGTH));
        // Verify that the encoder encode method was called with the correct message digest
        verify(encoder).encode(any(byte[].class));
    }

    private static Property mintProperty(String key) {
        return new Property(key);
    }
}
